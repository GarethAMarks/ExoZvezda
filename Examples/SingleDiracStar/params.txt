# Perturbations. With dirac_perturb = 1 the azimuthal index dirac_perturb_m selects the perturbation:
#   m = 0: the spherical perturbation (f, g) -> (1 + a exp(-(r - r0)^2 / w^2)) (f, g), with the metric re-solved, as in
#          Spherical_BS_Evolver (a = dirac_perturb_amp, w = dirac_perturb_spread, r0 = dirac_perturb_center);
#   m >= 1 (needs dirac_perturb_aspherical = 1): the aspherical perturbation of arXiv:2609.30021, Eq. (11): all matter
#          fields multiplied by exp(-2 delta cos(m phi)), delta = dirac_perturb_amp, phi = atan2(x_b, x_a) about the
#          star centre with (a, b) = dirac_perturb_axes (0 = x, 1 = y, 2 = z). The constraints are not re-solved.
# Two-axis perturbations: a second factor exp(-2 epsilon cos(n phi')) in an orthogonal plane, with n = dirac_perturb_n
# (0: none), epsilon = dirac_perturb_amp2 and the axes dirac_perturb_axes2 (sharing one axis with dirac_perturb_axes).
# An odd index is odd under the reflection of its first reference axis, so with the quadrant symmetry here (reflective
# y and z) odd m or n need x as the first reference axis. Single stars only.
# dirac_perturb_aspherical = 0 disables the aspherical perturbations (the perturbation is then always spherical).
# On the unstable branch (e.g. A = 0.30) m = 0 with a = +0.01 collapses to a black hole and a = -0.01 migrates to the
# stable branch (1D results with w = 3, r0 = 0).
dirac_perturb = 0
dirac_perturb_aspherical = 0
dirac_perturb_m = 0
dirac_perturb_amp = 0.01

dirac_perturb_center = 0.
dirac_perturb_axes = 0 1
dirac_perturb_n = 0
